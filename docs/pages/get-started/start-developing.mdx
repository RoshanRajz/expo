import React, { useState } from "react";
import { Input } from "@/components/ui/input";
import { Button } from "@/components/ui/button";
import { Card, CardContent } from "@/components/ui/card";
import { Label } from "@/components/ui/label";

const vehicleModels = [
  { model: "TVL-SL", price: 21000 },
  { model: "TVL-SL-PRO", price: 23500 },
  { model: "TVL-DL", price: 23500 },
  { model: "TVL-DL-PRO", price: 24500 },
  { model: "TVL-SCOOTIFY", price: 35000 },
  { model: "TVL-ALFABEAM", price: 37000 },
  { model: "TVL-ZIP", price: 49500 },
  { model: "TVL-POWER", price: 62000 }
];

export default function App() {
  const [vehicle, setVehicle] = useState(vehicleModels[0]);
  const [batteryType, setBatteryType] = useState("Lead Acid");
  const [batteryCount, setBatteryCount] = useState(1);
  const [leadType, setLeadType] = useState("12V*28Ah");
  const [lithVolt, setLithVolt] = useState(0);
  const [lithAmp, setLithAmp] = useState(0);
  const [chargerType, setChargerType] = useState("Lead Acid");
  const [chargerVolt, setChargerVolt] = useState(0);
  const [chargerAmp, setChargerAmp] = useState(0);
  const [totalCost, setTotalCost] = useState(0);

  const calculate = () => {
    let batteryCost = 0;
    let chargerCost = 0;

    if (batteryType === "Lead Acid") {
      const rate = leadType === "12V*28Ah" ? 2500 : 3200;
      const volt = leadType === "12V*28Ah" ? 12 : 12;
      batteryCost = rate * batteryCount;
    } else {
      const watt = lithVolt * lithAmp;
      batteryCost = watt * 12;
    }

    const chargerWatt = chargerVolt * chargerAmp;
    chargerCost = chargerWatt * (chargerType === "Lead Acid" ? 8 : 12);

    const total = vehicle.price + batteryCost + chargerCost;
    setTotalCost(total);
  };

  return (
    <div className="p-4 grid gap-4 max-w-xl mx-auto">
      <Card>
        <CardContent className="grid gap-4 p-4">
          <Label>Choose Vehicle Model</Label>
          <select onChange={(e) => setVehicle(vehicleModels[e.target.selectedIndex])}>
            {vehicleModels.map((v, i) => (
              <option key={i}>{v.model}</option>
            ))}
          </select>

          <Label>Battery Type</Label>
          <select onChange={(e) => setBatteryType(e.target.value)}>
            <option>Lead Acid</option>
            <option>Lithium</option>
          </select>

          {batteryType === "Lead Acid" ? (
            <>
              <Label>Choose Battery Spec</Label>
              <select onChange={(e) => setLeadType(e.target.value)}>
                <option value="12V*28Ah">12V*28Ah (₹2500)</option>
                <option value="12V*36Ah">12V*36Ah (₹3200)</option>
              </select>
              <Label>No. of Batteries</Label>
              <Input type="number" value={batteryCount} onChange={(e) => setBatteryCount(Number(e.target.value))} />
            </>
          ) : (
            <>
              <Label>Voltage (V)</Label>
              <Input type="number" value={lithVolt} onChange={(e) => setLithVolt(Number(e.target.value))} />
              <Label>Ampere (Ah)</Label>
              <Input type="number" value={lithAmp} onChange={(e) => setLithAmp(Number(e.target.value))} />
            </>
          )}

          <Label>Charger Type</Label>
          <select onChange={(e) => setChargerType(e.target.value)}>
            <option>Lead Acid</option>
            <option>Lithium</option>
          </select>

          <Label>Charger Voltage (V)</Label>
          <Input type="number" value={chargerVolt} onChange={(e) => setChargerVolt(Number(e.target.value))} />

          <Label>Charger Ampere (A)</Label>
          <Input type="number" value={chargerAmp} onChange={(e) => setChargerAmp(Number(e.target.value))} />

          <Button onClick={calculate}>Calculate Total Cost</Button>

          <h2 className="text-xl font-bold mt-4">Total Cost: ₹{totalCost}</h2>
        </CardContent>
      </Card>
    </div>
  );
}
